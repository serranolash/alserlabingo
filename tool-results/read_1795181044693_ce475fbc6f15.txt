   813→  const ranges = [
   814→    { start: 1, end: 15 },

   832→export function hasLine(grid: Grid, drawn: Set<number>): boolean {
   833→  // filas
   834→  for (let r = 0; r < 5; r++) {
   835→    if (grid[r].every((v) => v === null || drawn.has(v))) return true;
   836→  }
   837→  // columnas
   838→  for (let c = 0; c < 5; c++) {
   839→    let ok = true;
   840→    for (let r = 0; r < 5; r++) {
   841→      const v = grid[r][c];
   842→      if (v !== null && !drawn.has(v)) {
   843→        ok = false;
   844→        break;
   845→      }
   846→    }
   847→    if (ok) return true;
   848→  }
   849→  // diagonales
   850→  const d1 = [0, 1, 2, 3, 4].every((i) => grid[i][i] === null || drawn.has(grid[i][i]!));
   851→  const d2 = [0, 1, 2, 3, 4].every((i) => grid[i][4 - i] === null || drawn.has(grid[i][4 - i]!));
   852→  return d1 || d2;
   853→}

   860→export function hasFourCorners(grid: Grid, drawn: Set<number>): boolean {
   861→  const rows = grid.length;

   869→    grid[rows - 1][cols - 1],

   885→export function shuffle<T>(arr: T[]): T[] {
   886→  const a = [...arr];
   887→  for (let i = a.length - 1; i > 0; i--) {
   888→    const j = Math.floor(Math.random() * (i + 1));
   889→    [a[i], a[j]] = [a[j], a[i]];
   890→  }
   891→  return a;
